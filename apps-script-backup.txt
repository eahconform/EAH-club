/* ============================================================
   2. CHANGER L'URL GITHUB
============================================================ */

function updateSiteUrl(newUrl) {

  if (!newUrl) {
    throw new Error("URL obligatoire");
  }

  if (!newUrl.endsWith("/")) {
    newUrl += "/";
  }

  const ss = getSpreadsheet_();

  const paramSheet =
    ss.getSheetByName("PARAMETRES");

  const values =
    paramSheet
      .getDataRange()
      .getValues();

  for (let i = 1; i < values.length; i++) {

    if (values[i][0] === "SITE_URL") {

      paramSheet
        .getRange(i + 1, 2)
        .setValue(newUrl);
    }
  }

  const cartes =
    ss.getSheetByName("CARTES");

  const data =
    cartes
      .getRange(2, 1, cartes.getLastRow() - 1, 8)
      .getValues();

  for (let i = 0; i < data.length; i++) {

    const token = data[i][1];

    cartes
      .getRange(i + 2, 7)
      .setValue(
        newUrl +
        "?card=" +
        encodeURIComponent(token)
      );
  }

  return "URL mise à jour : " + newUrl;
}


/* ============================================================
   3. API WEB
============================================================ */

function doGet(e) {

  try {

    const action =
      String(e.parameter.action || "bootstrap");

    let result;

    switch (action) {

      case "bootstrap":

        result =
          getBootstrap_(
            e.parameter.card || ""
          );

        break;


      case "card":

        result =
          getCard_(
            e.parameter.card || ""
          );

        break;


      case "reservationStatus":

        result =
          getReservationStatus_(
            e.parameter.id || ""
          );

        break;


      case "health":

        result = {
          ok: true,
          message: "EAH API opérationnelle"
        };

        break;


      default:

        result = {
          ok: false,
          error: "Action inconnue"
        };

    }

    return jsonOutput_(
      result,
      e.parameter.callback
    );

  }

  catch (err) {

    return jsonOutput_({
      ok: false,
      error: err.message
    }, e.parameter.callback);

  }
}


/* ============================================================
   4. POST - ACTIVATION + RESERVATION
============================================================ */

function doPost(e) {

  try {

    const action =
      String(e.parameter.action || "");

    let result;

    if (action === "activate") {

      result =
        activateCard_(
          e.parameter.card,
          e.parameter.prenom,
          e.parameter.anniversaire,
          e.parameter.consentement
        );

    }

    else if (action === "reserve") {

      result =
        createReservation_({
          requestId: e.parameter.requestId,
          card: e.parameter.card,
          produitId: e.parameter.produitId,
          taille: e.parameter.taille,
          prenom: e.parameter.prenom,
          nom: e.parameter.nom,
          telephone: e.parameter.telephone,
          consentement: e.parameter.consentement
        });

    }

    else {

      result = {
        ok: false,
        error: "Action inconnue"
      };

    }

    return HtmlService
      .createHtmlOutput(
        "<html><body>OK</body></html>"
      );

  }

  catch (err) {

    return HtmlService
      .createHtmlOutput(
        "<html><body>ERREUR</body></html>"
      );

  }
}


/* ============================================================
   5. ACTIVATION CARTE
============================================================ */

function activateCard_(
  token,
  prenom,
  anniversaire,
  consentement
) {

  token = String(token || "").trim();
  prenom = String(prenom || "").trim();
  anniversaire = String(anniversaire || "").trim();

  if (!token) {
    throw new Error("Carte invalide");
  }

  if (
    prenom.length < 2 ||
    prenom.length > 40
  ) {
    throw new Error("Prénom invalide");
  }

  if (
    !/^(0?[1-9]|[12]\d|3[01])\/(0?[1-9]|1[0-2])$/
      .test(anniversaire)
  ) {
    throw new Error(
      "Anniversaire au format JJ/MM"
    );
  }

  if (consentement !== "true") {
    throw new Error(
      "Consentement obligatoire"
    );
  }

  const lock = LockService.getScriptLock();
  lock.waitLock(10000);

  try {

    const ss = getSpreadsheet_();
    const sheet = ss.getSheetByName("CARTES");

    const lastRow = sheet.getLastRow();

    const data = sheet
      .getRange(
        2,
        1,
        lastRow - 1,
        10
      )
      .getValues();

    for (let i = 0; i < data.length; i++) {

      if (
        String(data[i][COL.TOKEN - 1]) === token
      ) {

        const row = i + 2;

        const statut = String(
          data[i][COL.STATUT - 1] || "DISPONIBLE"
        ).trim();

        if (
          statut === "BLOQUÉE" ||
          statut === "PERDUE"
        ) {

          throw new Error(
            "Cette carte ne peut plus être activée."
          );
        }

        const alreadyActive =
          data[i][COL.ACTIVE - 1] === true;

        if (alreadyActive) {

          return {
            ok: true,
            alreadyActive: true
          };
        }

        sheet
          .getRange(row, COL.PRENOM)
          .setValue(prenom);

        sheet
          .getRange(row, COL.ANNIVERSAIRE)
          .setValue(anniversaire);

        sheet
          .getRange(row, COL.ACTIVE)
          .setValue(true);

        sheet
          .getRange(row, COL.DATE_ACTIVATION)
          .setValue(new Date());

        sheet
          .getRange(row, COL.STATUT)
          .setValue("ACTIVE");

        return {
          ok: true,
          activated: true
        };
      }
    }

    throw new Error("Carte inconnue");

  }

  finally {

    lock.releaseLock();
  }
}


/* ============================================================
   6. INFORMATIONS CARTE
============================================================ */

function getCard_(token) {

  token = String(token || "").trim();

  if (!token) {

    return {
      ok: true,
      exists: false,
      active: false
    };
  }

  const ss = getSpreadsheet_();
  const sheet = ss.getSheetByName("CARTES");

  const lastRow = sheet.getLastRow();

  if (lastRow < 2) {

    return {
      ok: true,
      exists: false,
      active: false
    };
  }

  const data = sheet
    .getRange(
      2,
      1,
      lastRow - 1,
      10
    )
    .getValues();

  for (let i = 0; i < data.length; i++) {

    if (
      String(data[i][COL.TOKEN - 1]) === token
    ) {

      const row = i + 2;

      const numero =
        data[i][COL.NUMERO - 1];

      const prenom =
        data[i][COL.PRENOM - 1];

      const anniversaire =
        data[i][COL.ANNIVERSAIRE - 1];

      const active =
        data[i][COL.ACTIVE - 1] === true;

      let statut = String(
        data[i][COL.STATUT - 1] || ""
      ).trim();

      // Pour compatibilité avec tes anciennes lignes
      if (!statut) {

        statut = active
          ? "ACTIVE"
          : "DISPONIBLE";
      }

      // Dernière utilisation
      sheet
        .getRange(
          row,
          COL.DERNIERE_UTILISATION
        )
        .setValue(new Date());


      // CARTE BLOQUÉE
      if (statut === "BLOQUÉE") {

        return {
          ok: true,
          exists: true,
          active: false,
          blocked: true,
          status: "BLOQUÉE",
          numero: numero
        };
      }


      // CARTE PERDUE
      if (statut === "PERDUE") {

        return {
          ok: true,
          exists: true,
          active: false,
          lost: true,
          status: "PERDUE",
          numero: numero
        };
      }


      // CARTE DISPONIBLE
      if (statut === "DISPONIBLE") {

        return {
          ok: true,
          exists: true,
          active: false,
          status: "DISPONIBLE",
          numero: numero
        };
      }


      // CARTE ACTIVE
      return {

        ok: true,

        exists: true,

        active: true,

        status: "ACTIVE",

        numero: numero,

        prenom: prenom,

        anniversaire: anniversaire,

        birthdayMonth:
          isBirthdayMonth_(
            anniversaire
          )
      };
    }
  }

  return {
    ok: true,
    exists: false,
    active: false
  };
}


/* ============================================================
   7. RESERVATION 24H
============================================================ */

function createReservation_(data) {

  if (
    String(data.consentement) !== "true"
  ) {
    throw new Error(
      "Consentement obligatoire"
    );
  }

  const produit =
    getProductById_(
      data.produitId
    );

  if (!produit) {
    throw new Error(
      "Produit inconnu"
    );
  }

  const prenom =
    String(data.prenom || "").trim();

  const nom =
    String(data.nom || "").trim();

  const telephone =
    String(data.telephone || "").trim();

  const taille =
    String(data.taille || "").trim();

  if (!prenom || !nom) {
    throw new Error(
      "Nom et prénom obligatoires"
    );
  }

  if (telephone.length < 8) {
    throw new Error(
      "Téléphone invalide"
    );
  }

  const now =
    new Date();

  const expiry =
    new Date(
      now.getTime() +
      24 * 60 * 60 * 1000
    );

  const requestId =
    String(
      data.requestId ||
      Utilities.getUuid()
    );

  let numeroCarte = "";

  if (data.card) {

    const card =
      getCard_(
        data.card
      );

    if (card.exists) {
      numeroCarte =
        card.numero || "";
    }
  }

  const ss =
    getSpreadsheet_();

  const sheet =
    ss.getSheetByName(
      "RESERVATIONS"
    );

  sheet.appendRow([

    requestId,

    now,

    expiry,

    "À CONFIRMER",

    produit.id,

    produit.nom,

    taille,

    prenom,

    nom,

    telephone,

    numeroCarte,

    data.card || "",

    "Réservation web 24h"

  ]);

  return {
    ok: true,
    id: requestId
  };
}

    ss.getSheetByName(
      "RESERVATIONS"
    );

  if (sheet.getLastRow() < 2) {

    return {
      ok: true,
      found: false
    };
  }

  const data =
    sheet
      .getRange(
        2,
        1,
        sheet.getLastRow() - 1,
        13
      )
      .getValues();

  for (let i = 0; i < data.length; i++) {

    if (
      String(data[i][0]) === String(id)
    ) {

      return {
        ok: true,
        found: true,
        status: data[i][3],
        expires: data[i][2]
      };
    }
  }

  return {
    ok: true,
    found: false
  };
}

function getSpreadsheet_() {

  const id =
    PropertiesService
      .getScriptProperties()
      .getProperty(
        "SPREADSHEET_ID"
      );

  if (!id) {

    throw new Error(
      "Exécute setupEAH() avant de déployer l'application."
    );
  }

  return SpreadsheetApp
    .openById(id);
}


function generateToken_() {

  return Utilities
    .getUuid()
    .replace(/-/g, "")
    .substring(0, 18)
    .toUpperCase();
}


function isBirthdayMonth_(birthday) {

  const parts =
    String(birthday || "")
      .split("/");

  if (parts.length !== 2) {
    return false;
  }

  const month =
    Number(parts[1]);

  return (
    month ===
    new Date().getMonth() + 1
  );
}


function jsonOutput_(
  object,
  callback
) {

  const json =
    JSON.stringify(object);

  if (callback) {

    const cb =
      String(callback)
        .replace(
          /[^a-zA-Z0-9_$\.]/g,
          ""
        );

    return ContentService
      .createTextOutput(
        cb +
        "(" +
        json +
        ");"
      )
      .setMimeType(
        ContentService
          .MimeType
          .JAVASCRIPT
      );
  }

  return ContentService
    .createTextOutput(json)
    .setMimeType(
      ContentService
        .MimeType
        .JSON
    );
}

}function ajouterCartes(nombre) {

  const ss = getSpreadsheet_();
  const sheet = ss.getSheetByName("CARTES");

  if (!sheet) {
    throw new Error("L'onglet CARTES est introuvable.");
  }

  nombre = Number(nombre);

  if (!nombre || nombre < 1) {
    throw new Error("Nombre de cartes invalide.");
  }

  // Récupération de l'URL actuelle du site
  const params = getParameters_();

  let siteUrl = String(params.SITE_URL || "").trim();

  if (!siteUrl) {
    throw new Error("SITE_URL n'est pas renseignée dans PARAMETRES.");
  }

  if (!siteUrl.endsWith("/")) {
    siteUrl += "/";
  }

  // Lecture des cartes existantes
  const lastRow = sheet.getLastRow();

  let existingData = [];

  if (lastRow >= 2) {
    existingData = sheet
      .getRange(2, 1, lastRow - 1, 8)
      .getValues();
  }

  // Trouver le plus grand numéro déjà utilisé
  let maxNumero = 0;

  const tokensExistants = new Set();

  existingData.forEach(row => {

    const numero = Number(row[0]);

    if (!isNaN(numero)) {
      maxNumero = Math.max(maxNumero, numero);
    }

    if (row[1]) {
      tokensExistants.add(String(row[1]));
    }

  });

  const nouvellesCartes = [];

  for (let i = 1; i <= nombre; i++) {

    const numero = maxNumero + i;

    let token;

    // Sécurité contre un doublon extrêmement improbable
    do {
      token = generateToken_();
    } while (tokensExistants.has(token));

    tokensExistants.add(token);

    const url =
      siteUrl +
      "?card=" +
      encodeURIComponent(token);

  nouvellesCartes.push([
  numero,          // NUMERO
  token,           // TOKEN
  "",              // PRENOM
  "",              // ANNIVERSAIRE
  false,           // ACTIVE
  "",              // DATE_ACTIVATION
  url,             // URL_NFC
  "",              // DERNIERE_UTILISATION
  false,           // PROGRAMMÉE
  "DISPONIBLE"     // STATUT
]);
  }

  const premiereNouvelleLigne = sheet.getLastRow() + 1;

  sheet
    .getRange(
      premiereNouvelleLigne,
      1,
      nouvellesCartes.length,
      10
    )
    .setValues(nouvellesCartes);
// Cases à cocher PROGRAMMÉE
sheet
  .getRange(
    premiereNouvelleLigne,
    COL.PROGRAMMEE,
    nouvellesCartes.length,
    1
  )
  .insertCheckboxes();


// Liste déroulante STATUT
const statutRule = SpreadsheetApp
  .newDataValidation()
  .requireValueInList(
    [
      "DISPONIBLE",
      "ACTIVE",
      "BLOQUÉE",
      "PERDUE"
    ],
    true
  )
  .setAllowInvalid(false)
  .build();

sheet
  .getRange(
    premiereNouvelleLigne,
    COL.STATUT,
    nouvellesCartes.length,
    1
  )
  .setDataValidation(statutRule);
  // Cases à cocher dans ACTIVE
  sheet
    .getRange(
      premiereNouvelleLigne,
      5,
      nouvellesCartes.length,
      1
    )
    .insertCheckboxes();

  // Ajustement visuel
  sheet.autoResizeColumns(1, 8);

  console.log(
    nombre +
    " cartes ajoutées. Numéros " +
    (maxNumero + 1) +
    " à " +
    (maxNumero + nombre)
  );

  return {
    ok: true,
    ajout: nombre,
    debut: maxNumero + 1,
    fin: maxNumero + nombre
  };
}

}function initialiserGestionCartes() {

  const ss = getSpreadsheet_();
  const sheet = ss.getSheetByName("CARTES");

  if (!sheet) {
    throw new Error("Onglet CARTES introuvable.");
  }

  const lastRow = sheet.getLastRow();

  if (lastRow < 2) {
    return;
  }

  // Titres
  sheet.getRange(1, COL.PROGRAMMEE).setValue("PROGRAMMÉE");
  sheet.getRange(1, COL.STATUT).setValue("STATUT");

  // Cases à cocher PROGRAMMÉE
  sheet
    .getRange(2, COL.PROGRAMMEE, lastRow - 1, 1)
    .insertCheckboxes();

  // Liste déroulante STATUT
  const rule = SpreadsheetApp
    .newDataValidation()
    .requireValueInList(
      [
        "DISPONIBLE",
        "ACTIVE",
        "BLOQUÉE",
        "PERDUE"
      ],
      true
    )
    .setAllowInvalid(false)
    .build();

  sheet
    .getRange(2, COL.STATUT, lastRow - 1, 1)
    .setDataValidation(rule);

  // Mettre DISPONIBLE uniquement si le statut est vide
  const data = sheet
    .getRange(2, COL.STATUT, lastRow - 1, 1)
    .getValues();

  for (let i = 0; i < data.length; i++) {

    if (!data[i][0]) {

      sheet
        .getRange(i + 2, COL.STATUT)
        .setValue("DISPONIBLE");
    }
  }

  console.log("Gestion des cartes initialisée.");
}function modifierStatutCarte_(
  numeroCarte,
  nouveauStatut
) {

  const statutsAutorises = [
    "DISPONIBLE",
    "ACTIVE",
    "BLOQUÉE",
    "PERDUE"
  ];

  if (
    !statutsAutorises.includes(
      nouveauStatut
    )
  ) {
    throw new Error("Statut invalide");
  }

  const ss = getSpreadsheet_();
  const sheet = ss.getSheetByName("CARTES");

  const data = sheet
    .getRange(
      2,
      1,
      sheet.getLastRow() - 1,
      10
    )
    .getValues();

  for (let i = 0; i < data.length; i++) {

    if (
      String(data[i][COL.NUMERO - 1]) ===
      String(numeroCarte)
    ) {

      const row = i + 2;

      sheet
        .getRange(row, COL.STATUT)
        .setValue(nouveauStatut);

      sheet
        .getRange(row, COL.ACTIVE)
        .setValue(
          nouveauStatut === "ACTIVE"
        );

      return {
        ok: true,
        numero: numeroCarte,
        statut: nouveauStatut
      };
    }
  }

  throw new Error("Carte introuvable");

}function remettreCarteDisponible(numeroCarte) {

  const ss = getSpreadsheet_();
  const sheet = ss.getSheetByName("CARTES");

  const data = sheet
    .getRange(
      2,
      1,
      sheet.getLastRow() - 1,
      10
    )
    .getValues();

  for (let i = 0; i < data.length; i++) {

    if (
      String(data[i][COL.NUMERO - 1]) ===
      String(numeroCarte)
    ) {

      const row = i + 2;

      // Suppression des données personnelles
      sheet
        .getRange(row, COL.PRENOM)
        .clearContent();

      sheet
        .getRange(row, COL.ANNIVERSAIRE)
        .clearContent();

      sheet
        .getRange(row, COL.DATE_ACTIVATION)
        .clearContent();

      // Désactivation
      sheet
        .getRange(row, COL.ACTIVE)
        .setValue(false);

      // Redevient disponible
      sheet
        .getRange(row, COL.STATUT)
        .setValue("DISPONIBLE");

      return {
        ok: true,
        numero: numeroCarte,
        statut: "DISPONIBLE"
      };
    }
  }

  throw new Error("Carte introuvable");
}function synchroniserStatutsCartes() {

  const ss = getSpreadsheet_();
  const sheet = ss.getSheetByName("CARTES");

  const lastRow = sheet.getLastRow();

  if (lastRow < 2) {
    return;
  }

  const data = sheet
    .getRange(2, 1, lastRow - 1, 10)
    .getValues();

  for (let i = 0; i < data.length; i++) {

    const row = i + 2;

    const active =
      data[i][COL.ACTIVE - 1] === true;

    const statut =
      String(
        data[i][COL.STATUT - 1] || ""
      ).trim();

    // Ne jamais toucher aux cartes bloquées ou perdues
    if (
      statut === "BLOQUÉE" ||
      statut === "PERDUE"
    ) {
      continue;
    }

    if (active) {

      sheet
        .getRange(row, COL.STATUT)
        .setValue("ACTIVE");

    } else if (!statut || statut === "ACTIVE") {

      sheet
        .getRange(row, COL.STATUT)
        .setValue("DISPONIBLE");
    }
  }

  console.log("Statuts synchronisés.");
}
